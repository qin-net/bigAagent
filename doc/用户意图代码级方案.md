```python
from pydantic import BaseModel, ConfigDict, Field, field_validator

NONE = "none"
SCHEMA = "1"
DIMS = (
    "fundamental",
    "technical",
    "sentiment",
    "macro",
    "decision",
    "tracking",
)
EFFECTS = ("this_run", "remember", "rerun", "remember_rerun")
MOMENTS = ("pre_run", "post_decision", "post_track")

TAG_ALIASES = {
    "#fundamental": "fundamental",
    "#基本面": "fundamental",
    "#technical": "technical",
    "#技术面": "technical",
    "#sentiment": "sentiment",
    "#情绪": "sentiment",
    "#macro": "macro",
    "#宏观": "macro",
    "#decision": "decision",
    "#决策": "decision",
    "#global": "decision",
    "#tracking": "tracking",
    "#追踪": "tracking",
    "#remember": "remember",
    "#记住": "remember",
    "#rerun": "rerun",
    "#重跑": "rerun",
}


def _none_str(v: str) -> str:
    s = v.strip()
    if s == "":
        raise ValueError("empty string not allowed; use none")
    return s


class UserUtterance(BaseModel):
    """一次发言事件。无原话。"""
    model_config = ConfigDict(extra="forbid")

    schema_version: str = SCHEMA
    utterance_id: str
    user_id: str
    moment: str
    effect: str
    tags: str                 # JSON list 的文本，无 tag 为 '["none"]'
    intent_id: str
    stock_code: str
    thesis_id: str
    run_id: str
    created_at: str

    @field_validator("moment")
    @classmethod
    def _moment(cls, v: str) -> str:
        if v not in MOMENTS:
            raise ValueError("bad moment")
        return v

    @field_validator("effect")
    @classmethod
    def _effect(cls, v: str) -> str:
        if v not in EFFECTS:
            raise ValueError("bad effect")
        return v

    @field_validator(
        "utterance_id", "user_id", "tags", "intent_id",
        "stock_code", "thesis_id", "run_id", "created_at",
    )
    @classmethod
    def _filled(cls, v: str) -> str:
        return _none_str(v)


class LlmIntentSlots(BaseModel):

    model_config = ConfigDict(extra="forbid")

    fundamental: str
    technical: str
    sentiment: str
    macro: str
    decision: str
    tracking: str
    not_evidence: str

    @field_validator("*")
    @classmethod
    def _slot(cls, v: str) -> str:

    schema_version: str = SCHEMA
    intent_id: str
    utterance_id: str
    effect: str
    tags: str

    technical: str
    sentiment: str
    macro: str
    decision: str
    tracking: str
    not_evidence: str

    @field_validator("effect")
    @classmethod
    def _effect(cls, v: str) -> str:
        if v not in EFFECTS:
            raise ValueError("bad effect")
        return v

    @field_validator(
        "intent_id", "utterance_id", "tags",
        "fundamental", "technical", "sentiment", "macro",
        "decision", "tracking", "not_evidence",
    )
    @classmethod
    def _filled(cls, v: str) -> str:
        return _none_str(v)


class UserPreference(BaseModel):
    model_config = ConfigDict(extra="forbid")

    schema_version: str = SCHEMA
    preference_id: str
    user_id: str

    trigger: str
    title: str
    statement: str
    source: str
    source_utterance_id: str
    source_run_id: str
    created_at: str
    updated_at: str

    @field_validator("*")
    @classmethod
    def _filled(cls, v: str) -> str:
        return _none_str(v)


class TagParse(BaseModel):

    tags: str                   # '["fundamental","remember"]' 或 '["none"]'
    effect: str
    body: str                   # 剥 tag 后；空则 "none"；仅内存，不入库

class ExpertTask(BaseModel):

    stock_code: str
    as_of: str
    output_schema_version: str
    instruction: str
    objective: str
    reason: str
    required_questions: list
    constraints: list
```

## 3. Tag 与抽取

`user_intent.py`：

```python
import json
import re
from uuid import uuid4

from .contracts import LLMMessage, LLMRequest
from .user_contracts import (
    DIMS, NONE, TAG_ALIASES, EFFECTS,
    LlmIntentSlots, TagParse, UserIntent,
)

TAG_RE = re.compile(r"(#[^\s#]+)", re.UNICODE)

EXTRACT_SYSTEM = """
You extract user intent for A-share research specialists.
Output JSON only. Keys: fundamental, technical, sentiment, macro,
decision, tracking, not_evidence.
Each value is a short Chinese task sentence, or the string none.
none means this slot has no user instruction.
Do not answer the stock question. Do not copy the user message verbatim.
Do not output null, missing keys, or empty strings.
If the text is unrelated, all slots are none.
""".strip()


def parse_tags(raw: str) -> TagParse:
    found = []
    dims = []
    remember = False
    rerun = False
    rest = raw
    for match in TAG_RE.finditer(raw):
        token = match.group(1).lower()
        # 中文 tag 不做 lower 破坏：再查原样
        key = match.group(1)
        mapped = TAG_ALIASES.get(key) or TAG_ALIASES.get(key.lower())
        if mapped is None:
            continue
        found.append(mapped)
        rest = rest.replace(match.group(1), " ", 1)
        if mapped == "remember":
            remember = True
        elif mapped == "rerun":
            rerun = True
        elif mapped in DIMS:
            dims.append(mapped)
    if remember and rerun:
        effect = "remember_rerun"
    elif remember:
        effect = "remember"
    elif rerun:
        effect = "rerun"
    else:
        effect = "this_run"
    body = rest.strip() or NONE
    tag_json = json.dumps(found or [NONE], ensure_ascii=False)
    dim_json = json.dumps(dims or [NONE], ensure_ascii=False)
    return TagParse(tags=tag_json, effect=effect, body=body, tagged_dims=dim_json)


def empty_slots() -> LlmIntentSlots:
    payload = {name: NONE for name in DIMS}
    payload["not_evidence"] = NONE
    return LlmIntentSlots.model_validate(payload)


async def extract_slots(llm, body: str, model: str) -> LlmIntentSlots:
    if body == NONE:
        return empty_slots()
    request = LLMRequest(
        model=model,
        messages=[
            LLMMessage(role="system", content=EXTRACT_SYSTEM),
            LLMMessage(role="user", content=body),
        ],
        thinking_enabled=False,
        response_format="json",
        max_tokens=1024,
    )
    try:
        response = await llm.complete(request)
        data = json.loads(response.content or "{}")
        return LlmIntentSlots.model_validate(data)
    except Exception:
        return empty_slots()


def apply_tag_authority(slots: LlmIntentSlots, parsed: TagParse) -> LlmIntentSlots:
    """被 tag 点名的维若仍是 none，保持 none（不把 body 原文写进槽）。
    点名只影响分流/remember 的 scope，不把原文当 statement。
    LLM 已填的非 none 保留。无 tag 时全听 LLM。
    """
    return slots


def build_intent(
    *,
    utterance_id: str,
    parsed: TagParse,
    slots: LlmIntentSlots,
) -> UserIntent:
    return UserIntent(
        intent_id=str(uuid4()),
        utterance_id=utterance_id,
        effect=parsed.effect,
        tags=parsed.tags,
        fundamental=slots.fundamental,
        technical=slots.technical,
        sentiment=slots.sentiment,
        macro=slots.macro,
        decision=slots.decision,
        tracking=slots.tracking,
        not_evidence=slots.not_evidence,
    )


def slot_of(intent: UserIntent, dim: str) -> str:
    return getattr(intent, dim)
```

说明：U1 **不把 body 程序写进槽**。tag 只定 effect 和 remember 的 scope。句子只来自 LLM；LLM 给 none 就是 none。避免原文落库。这和「tag 点名后程序用 body 补槽」相比更干净；你若坚持补槽，补的必须再压成一句且不得等于 body。

U1 对 `#rerun`：`effect` 仍记下，workflow **不调度重跑**（见 §6）。

```python
DEFAULT_INSTRUCTION = {
    "fundamental": (
        "Analyze this stock from the precomputed snapshot. "

    ),
    "technical": (
        "Analyze trend from precomputed indicators. "

    ),

MAX_SLOT_CHARS = 200
MAX_PREF = 8
MAX_PREF_CHARS = 80


def split_slot(slot: str) -> tuple[list, list]:
    if slot == NONE:
        return [], []
    text = slot[:MAX_SLOT_CHARS]
    if text.endswith("？") or text.endswith("?"):
        return [text], []
    return [], ["用户口径：" + text]


def build_expert_task(
    *,
    run_id: str,
    stock_code: str,
    as_of: str,
    dim: str,
    intent: UserIntent,
    preference_statements: list,
) -> str:
    questions, constraints = split_slot(slot_of(intent, dim))
    for stmt in preference_statements[:MAX_PREF]:
        constraints.append(stmt[:MAX_PREF_CHARS])
    task = {
        "run_id": run_id,
        "stock_code": stock_code,
        "as_of": as_of,
        "output_schema_version": "1",
        "instruction": DEFAULT_INSTRUCTION[dim],
        "objective": "完成本维分析",
        "reason": "首次研究",
        "required_questions": questions,
        "constraints": constraints,
    }
    return json.dumps(task, ensure_ascii=False)
```
